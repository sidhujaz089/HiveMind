def git_process_pids(read=None):
    """The pids of live `git` processes, read from /proc/<pid>/comm -- NOT `ps`, whose command column
    cuts at 80 columns without a terminal (library_checks ps-command-column; the worked example is
    check_stale_services.py). The program's own name is matched, so a python process whose arguments
    mention git, and a shell running `git status`, are not git and do not own the index (L-105).
    `read` -> {pid: comm} is injectable so a test runs no process table. -> sorted list[int]."""
    if read is None:
        procs = {}
        for entry in Path("/proc").iterdir():
            if entry.name.isdigit():
                try:
                    procs[int(entry.name)] = (entry / "comm").read_text().strip()
                except OSError:
                    procs[int(entry.name)] = ""      # exited between the listing and the read: not git
    else:
        procs = read()
    return sorted(int(pid) for pid, comm in procs.items() if comm == "git")
