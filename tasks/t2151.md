def tree_id_of(root, skip=('.git', 'chat', 'node_modules', '__pycache__', '.venv', 'build', '.gradle')):
    """Content identity of ANY directory tree: the repository, or a coder.py sandbox copy of it.

    tree_id() cannot be used on a sandbox: it asks git for the file list, and a sandbox excludes
    .git, so git answers with NOTHING and every sandbox in every dispatch hashes to one constant
    id -- which would make every recorded pass a hit on a tree that was never swept, the false
    green the task's note warns about. So this walks the tree instead of asking git, and it is the
    only implementation of the hashing rule.

    The same shape as tree_id(): relative path, then the file's bytes, into one sha256; a file that
    cannot be read hashes as b'<absent>' (not a match), and the paths this module writes itself
    (_SELF_WRITTEN) are skipped so a run's own output is never its input identity. Directories
    named in skip are not entered, so a sandbox tree hashes like the repo it was copied from
    instead of hashing a multi-gigabyte .venv. A directory with no files hashes to the empty-input
    digest: that is a distinct, stable id, not a collision with a real tree.

    Returns a 16-hex id, the same width as tree_id() and suite_key().
    """
    root = Path(root)
    h = hashlib.sha256()
    for dirpath, dirnames, filenames in os.walk(str(root)):
        dirnames[:] = sorted(d for d in dirnames if d not in skip)
        for fn in sorted(filenames):
            p = Path(dirpath) / fn
            rel = p.relative_to(root).as_posix()
            if rel in _SELF_WRITTEN:
                continue
            h.update(rel.encode())
            try:
                h.update(hashlib.sha256(p.read_bytes()).digest())
            except OSError:
                h.update(b'<absent>')
    return h.hexdigest()[:16]
